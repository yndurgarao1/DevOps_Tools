#common steps for master node aand worker node 

#Ensure your system has the latest package lists before installation.
root@ip-4-8-4-11:/home/ubuntu# apt update -y

#Load Kernel Modules ( Networking modules are required for Kubernetes networking. )
root@ip-4-8-4-11:/home/ubuntu# cat <<EOF | sudo tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

#ensures that the kernel supports OverlayFS, which is critical for Docker and Kubernetes to handle container image layers efficiently.
root@ip-4-8-4-11:/home/ubuntu# sudo modprobe overlay ==> 

#enables iptables to filter bridged traffic, which is essential for Kubernetes networking and CNI plugins (like Calico, Flannel, etc.) to function correctly.
root@ip-4-8-4-11:/home/ubuntu# sudo modprobe br_netfilter

#Configure Sysctl Parameters - Enable IP forwarding and bridge traffic for Kubernetes networking.
root@ip-4-8-4-11:/home/ubuntu# cat <<EOF | sudo tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF


# Apply sysctl params without reboot
root@ip-4-8-4-11:/home/ubuntu# sudo sysctl --system

# Verify that the br_netfilter, overlay modules are loaded by running the following commands:
root@ip-4-8-4-11:/home/ubuntu# lsmod | grep br_netfilter
root@ip-4-8-4-11:/home/ubuntu# lsmod | grep overlay


# Verify that the net.bridge.bridge-nf-call-iptables, net.bridge.bridge-nf-call-ip6tables, and net.ipv4.ip_forward system variables are set to 1 in your sysctl config by running the following command:
root@ip-4-8-4-11:/home/ubuntu# sysctl net.bridge.bridge-nf-call-iptables net.bridge.bridge-nf-call-ip6tables net.ipv4.ip_forward
