##Generating TLS Certificate in one of the ETCD EC2 Instance
For Generating TLS Certificate we are using cfssl from cloudware 
URL: https://github.com/cloudflare/cfssl/releases
we need cfssl and cfssljson 

https://github.com/cloudflare/cfssl/releases/download/v1.7.1/cfssl_1.7.1_linux_amd64
https://github.com/cloudflare/cfssl/releases/download/v1.7.1/cfssljson_1.7.1_linux_amd64


[root@etcd-node1 ec2-user]# curl -s -L -o /usr/bin/cfssl https://github.com/cloudflare/cfssl/releases/download/v1.7.1/cfssl_1.7.1_linux_amd64
[root@etcd-node1 ec2-user]# chmod +x /usr/bin/cfssl
[root@etcd-node1 ec2-user]# curl -s -L -o /usr/bin/cfssljson https://github.com/cloudflare/cfssl/releases/download/v1.7.1/cfssljson_1.7.1_linux_amd64

##etcd as a service ( Perform in all etcd nodes )

[root@etcd-node1 ec2-user]# cd /etc/systemd/system
[root@etcd-node1 ec2-user]# vi /etcd.service
[Unit]
Description=etcd server

[Service]
Type=notify
User=etcd
EnvironmentFile=/etc/etcd_config
ExecStart=/usr/bin/etcd
Restart=on-failure
RestartSec=5


[Install]
WantedBy=multi-user.target
